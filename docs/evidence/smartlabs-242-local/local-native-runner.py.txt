def run(name,args,env=None,stdin=None,timeout=600):
    e=os.environ.copy();e.pop('NODE_OPTIONS',None)
    if env:e.update(env)
    p=subprocess.run(args,cwd=ROOT,env=e,input=stdin,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=timeout)
    raw=p.stdout
    (OUT/(name+'.log')).write_bytes(raw+b'\nEXIT='+str(p.returncode).encode()+b'\n')
    receipt={'command':args,'exit':p.returncode,'sha256':hashlib.sha256(raw).hexdigest()}
    (OUT/(name+'.json')).write_text(json.dumps(receipt,indent=2)+'\n')
    print(name,'EXIT',p.returncode,raw.decode(errors='replace')[-1400:])
    return p
