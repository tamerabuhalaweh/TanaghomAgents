import assert from 'node:assert/strict';import test from 'node:test';import {readFileSync,writeFileSync} from 'node:fs';import {spawnSync} from 'node:child_process';
import {validateMp4} from '../tanaghom-p5-decision/packages/creative-runtime/render/mp4.mjs';
const fixture=readFileSync(new URL('../tanaghom-p5-decision/evaluation/creative-talking-head-v1/fixtures/encoded.mp4',import.meta.url));
const opts={expectedDurationSec:5,allowedCodecs:['avc1'],requireAudio:true};
function boxes(b,s=0,e=b.length){let out=[];for(let p=s;p+8<=e;){const size=b.readUInt32BE(p);assert.ok(size>=8&&p+size<=e);out.push({type:b.toString('latin1',p+4,p+8),start:p,end:p+size});p+=size;}return out;}
function find(b,t,s=0,e=b.length){for(const x of boxes(b,s,e)){if(x.type===t)return x;if(['moov','trak','mdia','minf','stbl'].includes(x.type)){const v=find(b,t,x.start+8,x.end);if(v)return v;}}}
test('positive encoded fixture actual structural metadata and ffmpeg decode',()=>{const m=validateMp4(fixture,opts);assert.equal(m.codec,'avc1');assert.equal(m.audio.codec,'mp4a');const r=spawnSync('ffmpeg',['-v','error','-i',new URL('../tanaghom-p5-decision/evaluation/creative-talking-head-v1/fixtures/encoded.mp4',import.meta.url).pathname.replace(/^\/(\w:)/,'$1'),'-f','null','-'],{encoding:'utf8'});console.log(JSON.stringify({metadata:m,decode_exit:r.status,decode_error:r.stderr}));assert.equal(r.status,0);});
test('nonexistent active sample description must fail structural admission',()=>{const b=Buffer.from(fixture),stsd=find(b,'stsd'),stsc=find(b,'stsc');const entries=boxes(b,stsd.start+16,stsd.end);assert.equal(entries.length,1);assert.equal(entries[0].type,'avc1');b.writeUInt32BE(2,stsd.start+12);for(let i=0;i<b.readUInt32BE(stsc.start+12);i++)b.writeUInt32BE(2,stsc.start+24+i*12);const target=new URL('nonexistent-description.mp4',import.meta.url);writeFileSync(target,b);let accepted=false,meta,error;try{meta=validateMp4(b,opts);accepted=true;}catch(e){error=e.message;}const r=spawnSync('ffmpeg',['-v','error','-i',target.pathname.replace(/^\/(\w:)/,'$1'),'-f','null','-'],{encoding:'utf8'});console.log(JSON.stringify({actual_entries:entries.length,declared_entries:2,active_description:2,accepted,meta,error,decode_exit:r.status,decode_error:r.stderr}));assert.equal(accepted,false,'a sample map referring to missing stsd entry must not be admitted');});
